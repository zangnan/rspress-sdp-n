    #事件 & API

    预计阅读时间: 5 分钟
    For your Agent
    一键创建 Sheet.模块

    复制这段 Prompt 发送给你的 AI Agent，将他作为开发基础列表模块的开发指南。

    按照基础列表事件 & API文档 http://localhost:3000/frontdocv2/component/basegrid/api_event.html 创建一个基础列表模块。 1. 引入并使用 DDBaseGrid 组件，通过 config 配置 api_grid_query 等基础属性。 2. 在 config.grid_columns 中定义列表列，设置 title、dataIndex、width 等基础字段。 3. 按数据类型为列配置 render，支持数值、日期、码表、图标等格式化方式。 4. 按需配置列的 filter、codeTable 及 func、actionRef、queryParam、onRow 等扩展能力。

    #config.grid_columns.render

    #格式化数值

    render: { type: 'number', format: '0,0.0000' }
render: { type: 'number', format: '0,0.00' }

    #格式化时间

    render: { type: 'date', format: 'YYYY-MM-DD' }
render: { type: 'date', format: 'YYYY-MM-DD HH:mm:ss' }

    #前端格式化码表

    • 也可以利用 codeTable 属性格式化码表
    • 可结合 格式化图标类 配套使用
    render: {
  map: [
    { value: 'Y', text: '检查通过' },
    { value: 'N', text: '检查未通过' },
    { value: '', text: '未检查' },
  ]
}

    #格式化图标类

    // color可选: default|success|processing|error|warning
render: {
  // Tag
  tag: [
    { value: '0139', color: 'default' },
    { value: '0130', color: 'error' },
    { value: '', color: 'processing' },
  ],
  // 图标
  icon: [
    { value: '0139', icon: 'CheckOutlined', color: 'default' },
    { value: '0130', icon: 'CoffeeOutlined', color: 'error' },
    { value: '', icon: 'RobotOutlined', color: 'processing' },
  ],
  // 状态点
  circle: [
    { value: '0139', color: 'default' },
    { value: '0130', color: 'error' },
    { value: '', color: 'processing' },
  ]
}

    #自定义格式化

    • 需要结合func属性配套使用
    • 参照示例 后端分页 - 复杂示例
    render: { func: 'renderExampleXh4' }

    #config.grid_columns.codeTable

    • 也可以利用 config.grid_columns.render 前端格式化码表 属性格式化码表
    • 可结合 格式化图标类 配套使用
    codeTable: ['code_dm_ckts_hgjldw', '01']

    #config.grid_columns.filter

    #多选

    // 基于码表取值
filter: {
  type: 'Select',
  codeTable: ['code_dm_ckts_hgjgfs', '01'],
  showCode: true,
}
// 基于前端取值
filter: {
  type: 'Select',
  list: [
    { text: '已结关', value: 'yjg' },
    { text: '未结关', value: 'wjg' },
  ],
}

    #单选

    // 基于码表取值
filter: {
  type: 'Radio',
  codeTable: ['code_dm_ckts_hgzfjsfs', '01'],
},
// 基于前端取值
filter: {
  type: 'Radio',
  list: [
    { text: '全部', value: 'qb' },
    { text: '通过', value: 'tg' },
    { text: '驳回', value: 'bh' },
  ],
},
